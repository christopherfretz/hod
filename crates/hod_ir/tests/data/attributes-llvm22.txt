# One line per attribute: <form> <name> <positions>.
#   form: the TableGen class without its "Attr" suffix, in lower case
#         (enum, int, type, constantrange, constantrangelist: keywords;
#          strbool, complexstr: string attributes).
#   positions: fn, param, ret as the definition lists them; "-" for none.
# Regenerate with (POSIX sh, from the directory holding Attributes.td):
#   tr '\n' ' ' < Attributes.td \
#   | grep -oE 'def +[A-Za-z0-9_]+ *: *(EnumAttr|IntAttr|TypeAttr|ConstantRangeAttr|ConstantRangeListAttr|StrBoolAttr|ComplexStrAttr) *< *"[^"]+"[^>]*>' \
#   | sed -E 's/def +[A-Za-z0-9_]+ *: *([A-Za-z]+)Attr *< *"([^"]+)"(.*)/\1 \2 \3/' \
#   | awk '{ form = tolower($1); name = $2; pos = "";
#            if ($0 ~ /FnAttr/) pos = pos "fn,"; if ($0 ~ /ParamAttr/) pos = pos "param,"; if ($0 ~ /RetAttr/) pos = pos "ret,";
#            if (pos == "") pos = "-,"; print form, name, substr(pos, 1, length(pos) - 1) }' \
#   | LC_ALL=C sort -k2,2
